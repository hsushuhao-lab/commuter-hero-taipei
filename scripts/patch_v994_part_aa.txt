from pathlib import Path
import base64
import re


def replace_once(path, old, new, label):
    p = Path(path)
    text = p.read_text(encoding='utf-8')
    if old not in text:
        raise SystemExit(f'FAILED {label}: target not found in {path}')
    text = text.replace(old, new, 1)
    p.write_text(text, encoding='utf-8')
    print(f'PASS {label}')


def regex_replace_once(path, pattern, repl, label, flags=re.S):
    p = Path(path)
    text = p.read_text(encoding='utf-8')
    new, n = re.subn(pattern, repl, text, count=1, flags=flags)
    if n != 1:
        raise SystemExit(f'FAILED {label}: matches={n} in {path}')
    p.write_text(new, encoding='utf-8')
    print(f'PASS {label}')

# ------------------------------------------------------------------
chunk_paths = sorted(Path('scripts').glob('menu_keyart_v994_*.b64'))
missing = [str(p) for p in chunk_paths if not p.exists()]
if missing:
    raise SystemExit(f'FAILED menu art payload missing: {missing}')
payload = ''.join(p.read_text(encoding='ascii').strip() for p in chunk_paths)
art_bytes = base64.b64decode(payload, validate=True)
asset_path = Path('assets/menu_keyart_v994.jpg')
asset_path.write_bytes(art_bytes)
print(f'PASS menu art decoded: {asset_path} ({len(art_bytes)} bytes)')
